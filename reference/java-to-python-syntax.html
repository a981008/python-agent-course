装饰器（Agent 代码高频模式）

import functools

def log_call(fn):
    @functools.wraps(fn)              # 保留原函数的元信息
    def wrapper(*args, **kwargs):
        print(f"调用 {fn.__name__}, 参数 {args}")
        return fn(*args, **kwargs)
    return wrapper

@log_call                     # 等价于 pay = log_call(pay)
def pay(order_id: str) -> bool:
    return True

# 读法：@ 是语法糖。LangChain 的 @tool、FastAPI 的 @app.get
# 本质都是"用装饰器注册函数"

# ── 带参数的装饰器 = 装饰器工厂：先收参数、再收函数，三层结构 ──
# 你见过的 @dataclass(frozen=True)、@app.get("/pay")、@retry(times=3) 都属于这类
def retry(times: int):                    # 第 1 层：装饰器工厂，收装饰器参数
    def decorator(fn):                    # 第 2 层：普通装饰器，收函数
        @functools.wraps(fn)
        def wrapper(*args, **kwargs):     # 第 3 层：包装器，真正执行
            for i in range(times):
                try:
                    return fn(*args, **kwargs)
                except Exception:
                    if i == times - 1:
                        raise             # 用尽次数向上抛（保留原异常）
        return wrapper
    return decorator

@retry(times=3)            # 等价于 pay = retry(times=3)(pay)：先调用工厂得到装饰器
def pay(order_id: str) -> bool: ...

# 写带参装饰器记法：套三层，参数一层比一层靠里；@log_call 是零参数特例（只有两层）

类与 dataclass

class Payment:
    def __init__(self, order_id: str, amount: float):   # → 构造器
        self.order_id = order_id        # 实例属性（无需预先声明字段）
        self.amount = amount

    def is_paid(self) -> bool:          # 实例方法：第一个参数是 self
        return self.status == "PAID"

    @staticmethod
    def validate(amount: float) -> bool:    # → static 方法
        return amount > 0

    @property
    def yuan(self) -> float:            # 像字段一样访问：p.yuan
        return self.amount / 100        # （计算属性，无需 get 前缀）

class Refund(Payment):                  # → extends
    def __init__(self, order_id, amount, reason):
        super().__init__(order_id, amount)
        self.reason = reason

# for 循环的本质：iter() 拿迭代器，反复 next() 直到 StopIteration

# 自定义迭代器：实现 __iter__ / __next__（→ Java 的 Iterable/Iterator）
class Countdown:
    def __init__(self, n): self.n = n
    def __iter__(self): return self          # 迭代器返回自身
    def __next__(self):
        if self.n <= 0: raise StopIteration
        self.n -= 1
        return self.n + 1

# yield from：委托给子生成器（递归/组合流式管道）
def tree(node):
    yield node.name
    for child in node.children:
        yield from tree(child)      # 展开子生成器的所有值

# 生成器的双向通信：send() 把值送回 yield 表达式
def echo():
    while True:
        got = yield          # yield 的值 = send() 传入的值
        print("收到:", got)

作用域与闭包：LEGB 与 nonlocal（官方教程 §4.7 / §9.2）

# LEGB 查找顺序：Local → Enclosing → Global → Builtins
x = "global"
def outer():
    x = "enclosing"
    def inner():
        x = "local"          # 就近遮蔽，不会改外层
        return x
    return inner()

# 闭包：内部函数记住外层变量（→ Java 匿名类只能捕获 final）
def make_counter():
    count = 0
    def inc():
        nonlocal count       # 没有它，赋值会创建新的局部变量！
        count += 1
        return count
    return inc
c = make_counter()
c(); c()                     # 1, 2 —— 状态保存在闭包里

# global：函数内给模块级变量赋值必须声明（只读不需要）
# 常见坑：循环变量被闭包共享
funcs = [lambda: i for i in range(3)]
[f() for f in funcs]         # [2, 2, 2]！三处都引用同一个 i
funcs = [lambda i=i: i for i in range(3)]   # 用默认参数冻结当前值

对象模型：id / 可哈希 / 引用计数（官方教程 §3.1）

# 一切皆对象；变量只是名字贴在对象上（→ Java 引用语义，但无基本类型/包装类型之分）
a = [1, 2]; b = a
b.append(3); a               # [1, 2, 3] —— 同一个对象
id(a) == id(b)               # True：对象身份（→ System.identityHashCode）
import copy
c = copy.deepcopy(a)         # 深拷贝（Java: 序列化克隆）

# 可哈希才能做 dict 键 / 进 set：数字、字符串、tuple（含不可变元素）可以；list/dict/set 不行
{(1, 2): "point"}            # tuple 当键 OK
# 小整数与短字符串有缓存（-5..256），is 判断可能碰巧为 True —— 别依赖，判等一律 ==

# 引用计数 + 循环垃圾回收（→ JVM GC，但 CPython 主靠引用计数，对象无引用立刻回收）
import sys
sys.getrefcount(a)           # 比直觉多 1（参数引用）
del a                        # 只删名字，不删对象（对象无引用才销毁）

# __slots__：锁定属性集合，省内存、禁动态加属性（→ Java 的字段固定声明）
class Point:
    __slots__ = ("x", "y")

私有变量与 name mangling（官方教程 §9.6）

# Python 没有private/protected 关键字，全靠约定 + 改名机制
class Account:
    def __init__(self):
        self.owner = "公开"          # → public
        self._balance = 0            # 单下划线：约定"内部使用"，外部仍能访问
        self.__pin = "1234"          # 双下划线：触发 name mangling

acct = Account()
acct._balance          # 能访问，但 IDE/约定都会警告你（→ @VisibleForTesting 的味道）
acct.__pin             # AttributeError！
acct._Account__pin     # mangling 后的真实名字：_类名__属性（不要这样写，知道即可）

# 双下划线防的是"子类无意覆盖"，不是真正的访问控制
# 属性统一走 @property（→ getter/setter，但调用方写法不变，见"类与 dataclass"节）

from enum import Enum, auto

# → Java 的 enum，但成员是类属性，值任意
class Status(Enum):
    PENDING = auto()         # auto() 自动编号
    PAID = "paid"
    SHIPPED = "shipped"

Status.PAID                 # <Status.PAID: 'paid'>
Status.PAID.value           # 'paid'
Status("paid")              # 按值反查（Java: valueOf 语义，查不到抛 ValueError）
Status.PAID.name            # 'PAID'
list(Status)                # 可迭代所有成员

class Level(IntEnum):
    LOW = 1
    HIGH = 2
Level.HIGH > Level.LOW      # True
