装饰器（Agent 代码高频模式）

import functools

def log_call(fn):
    @functools.wraps(fn)              # 保留原函数的元信息
    def wrapper(*args, **kwargs):
        print(f"调用 {fn.__name__}, 参数 {args}")
        return fn(*args, **kwargs)
    return wrapper

@log_call                     # 等价于 pay = log_call(pay)
def pay(order_id: str) -> bool:
    return True

# 读法：@ 是语法糖。LangChain 的 @tool、FastAPI 的 @app.get
# 本质都是"用装饰器注册函数"

# ── 带参数的装饰器 = 装饰器工厂：先收参数、再收函数，三层结构 ──
# 你见过的 @dataclass(frozen=True)、@app.get("/pay")、@retry(times=3) 都属于这类
def retry(times: int):                    # 第 1 层：装饰器工厂，收装饰器参数
    def decorator(fn):                    # 第 2 层：普通装饰器，收函数
        @functools.wraps(fn)
        def wrapper(*args, **kwargs):     # 第 3 层：包装器，真正执行
            for i in range(times):
                try:
                    return fn(*args, **kwargs)
                except Exception:
                    if i == times - 1:
                        raise             # 用尽次数向上抛（保留原异常）
        return wrapper
    return decorator

@retry(times=3)            # 等价于 pay = retry(times=3)(pay)：先调用工厂得到装饰器
def pay(order_id: str) -> bool: ...

# 写带参装饰器记法：套三层，参数一层比一层靠里；@log_call 是零参数特例（只有两层）

类与 dataclass

from decimal import Decimal

class Payment:
    currency = "CNY"               # 类变量：类似 Java static 字段

    def __init__(self, order_id: str, amount: float):   # → 构造器
        self.order_id = order_id        # 实例属性（无需预先声明字段）
        self.amount = amount

    @classmethod
    def from_yuan(cls, order_id: str, yuan: str):       # cls 收到实际子类
        amount = int(Decimal(yuan) * 100)
        return cls(order_id, amount)     # 工厂方法；子类调用会构造子类

    def is_paid(self) -> bool:          # 实例方法：第一个参数是 self
        return self.status == "PAID"

    @staticmethod
    def validate(amount: float) -> bool:    # → static 方法
        return amount > 0

    @property
    def yuan(self) -> float:            # 像字段一样访问：p.yuan
        return self.amount / 100        # （计算属性，无需 get 前缀）

class Refund(Payment):                  # → extends
    def __init__(self, order_id, amount, reason):
        super().__init__(order_id, amount)
        self.reason = reason

# ExceptionGroup / except*：把多个独立异常打包，再按类型分别处理（3.11+）
def parse_all(items):
    errors = []
    for item in items:
        try:
            int(item)
        except (TypeError, ValueError) as exc:
            errors.append(exc)
    if errors:
        raise ExceptionGroup("部分记录解析失败", errors)

try:
    parse_all(["1", "x", None])
except* ValueError as group:
    for exc in group.exceptions:
        exc.add_note("请检查该记录的值")

# 序列切片 seq[start:stop:step] —— Java 只能 list.subList()，取尾、步长都要手写循环

class Countdown:
    def __init__(self, n): self.n = n
    def __iter__(self): return self          # 迭代器返回自身
    def __next__(self):
        if self.n <= 0: raise StopIteration
        self.n -= 1
        return self.n + 1

class Bag:
    def __init__(self, items): self.items = list(items)
    def __len__(self): return len(self.items)          # len(bag)
    def __getitem__(self, i): return self.items[i]     # bag[i] / bag[1:3]
    def __contains__(self, x): return x in self.items  # x in bag
    def __iter__(self): return iter(self.items)        # for x in bag
    def __call__(self, x): return x in self             # bag(x)
    def __repr__(self): return f"Bag({self.items!r})"

class Money:
    def __init__(self, cents): self.cents = cents
    def __add__(self, other):                           # money + money
        return Money(self.cents + other.cents)
    def __radd__(self, other):                          # sum([...]) 从 0 开始
        if other == 0: return self
        return self.__add__(other)
    def __eq__(self, other):
        return isinstance(other, Money) and self.cents == other.cents
    def __hash__(self):                                 # 自定义 __eq__ 后进 set 要补
        return hash(self.cents)

异步进阶（第 22 课）

import asyncio

async def work(x):
    await asyncio.sleep(0.1)
    return x

async def main():
    # TaskGroup：结构化并发；一个任务失败会取消同组其他任务
    async with asyncio.TaskGroup() as tg:
        t1 = tg.create_task(work(1))
        t2 = tg.create_task(work(2))
    print(t1.result(), t2.result())

    # 总超时：3.11+ 的上下文管理器，超时会取消内部任务
    try:
        async with asyncio.timeout(1):
            await work(3)
    except TimeoutError:
        print("timeout")

    # 并发上限：Semaphore 限制同时进入临界区的协程数
    sem = asyncio.Semaphore(3)
    async def limited(i):
        async with sem:
            return await work(i)
    await asyncio.gather(*[limited(i) for i in range(10)])

    # 取消：协作式，在 await 点收到 CancelledError；finally 做清理
    task = asyncio.create_task(work(4))
    task.cancel()
    try:
        await task
    except asyncio.CancelledError:
        print("cancelled")

    # 必须完成的收尾：shield 保护不被外层取消
    await asyncio.shield(work(5))

    # 异步迭代与异步上下文管理器
    async for chunk in stream():
        ...
    async with session() as s:
        ...

    # 阻塞调用丢到线程池，避免卡住事件循环
    data = await asyncio.to_thread(blocking_io, "x")

async def stream():
    for i in range(3):
        yield i

asyncio.run(main())
